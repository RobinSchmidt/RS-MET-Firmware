```c
#define SUPERSAW_VOICES 7

// Klassische JP-8000-inspirierte Verstimmungs-Offsets (symmetrisch um 0)
// Werte sind relative Frequenz-Multiplikatoren skaliert auf Q16 (65536 = 1.0)
static const int32_t kDetuneOffsets[SUPERSAW_VOICES] = {
    -7210, -4064, -1280, 0, 1280, 4064, 7210
};

// Lautstärke-Gewichtung pro Oszillator (Center etwas lauter, Seiten leicht gedämpft)
// Summe ist ~32768 (Q15), damit der Mix nicht übersteuert
static const int32_t kOscGainsQ15[SUPERSAW_VOICES] = {
    3900, 4300, 4700, 6968, 4700, 4300, 3900
};

typedef struct {
    uint32_t phase[SUPERSAW_VOICES];
    uint32_t inc[SUPERSAW_VOICES];
} SupersawVoice;

// Schneller Q15-PolyBLEP (korrigiert die Sprungstelle bei Phase 0 / Wrap)
// t  = aktuelle Phase (uint32_t)
// dt = Phasen-Inkrement (uint32_t)
// Rückgabe: Korrekturwert im Bereich [-32768 .. +32768] (Q15)
static inline int32_t polyblep_q15(uint32_t t, uint32_t dt) {
    if (dt == 0) return 0;

    // Direkt NACH dem Phasensprung (t < dt)
    if (t < dt) {
        // u = t / dt skaliert auf Q15 (0 .. 32768)
        int32_t u = (int32_t)(((uint64_t)t << 15) / dt);
        // Polynom: 2*u - u^2 - 1 (in Q15)
        int32_t u2 = (u * u) >> 15;
        return (u << 1) - u2 - 32768;
    }
    // Direkt VOR dem Phasensprung (t > 0xFFFFFFFF - dt)
    uint32_t dist_to_end = ~t; // entspricht (0xFFFFFFFF - t)
    if (dist_to_end < dt) {
        // u = -(dist_to_end / dt) skaliert auf Q15 (-32768 .. 0)
        int32_t u = -(int32_t)(((uint64_t)dist_to_end << 15) / dt);
        // Polynom: u^2 + 2*u + 1 (in Q15)
        int32_t u2 = (u * u) >> 15;
        return u2 + (u << 1) + 32768;
    }
    return 0;
}

void supersaw_init(SupersawVoice *v) {
    for (int i = 0; i < SUPERSAW_VOICES; i++) {
        v->phase[i] = (uint32_t)i * 0x24924924U; // Gleichmäßig über 360° verteilt
        v->inc[i] = 0;
    }
}

// detune_q15: Detune-Regler von 0 (unisono) bis 32768 (maximaler Rave-Spread)

    for (int i = 0; i < SUPERSAW_VOICES; i++) {
        // Offset = base_inc * (kDetuneOffsets[i] * detune_q15)
        int64_t mod = (int64_t)kDetuneOffsets[i] * detune_q15; // Q31
        int64_t delta = ((int64_t)base_inc * mod) >> 31;
        v->inc[i] = (uint32_t)((int64_t)base_inc + delta);
    }
}

// Rendert einen Audio-Block (Stereo, Q15-Output)
// stereo_spread_q15: 0 = Mono, 32768 = voller Stereo-Spread
void supersaw_process(SupersawVoice *v, int16_t *out_l, int16_t *out_r,
                      int num_samples, int32_t stereo_spread_q15) {
    // Vorberechnung der Stereo-Pans für die 7 Oszillatoren (Außenseiten L/R, Mitte Center)
    int32_t pan_l[SUPERSAW_VOICES];
    int32_t pan_r[SUPERSAW_VOICES];
    const int32_t base_pan[SUPERSAW_VOICES] = {-32768, 24000, -14000, 0, 14000, -24000, 32768};

    for (int i = 0; i < SUPERSAW_VOICES; i++) {
        int32_t pan = (base_pan[i] * stereo_spread_q15) >> 15; // -32768 (L) .. +32768 (R)
        pan_l[i] = (32768 - pan) >> 1; // 0 .. 32768
        pan_r[i] = (32768 + pan) >> 1; // 0 .. 32768
    }

    for (int n = 0; n < num_samples; n++) {
        int32_t acc_l = 0;
        int32_t acc_r = 0;

        for (int i = 0; i < SUPERSAW_VOICES; i++) {
            uint32_t p = v->phase[i];
            uint32_t dt = v->inc[i];

            // Naiver Sägezahn aus den oberen 16 Bit der Phase: [-32768 .. +32767]
            int32_t saw = (int32_t)(p >> 16) - 32768;

            // Bandlimiting an der Flanke abziehen
            saw -= polyblep_q15(p, dt);

            // Mit Oszillator-Gain gewichten (Q15)
            int32_t weighted = (saw * kOscGainsQ15[i]) >> 15;

            // Auf linken und rechten Kanal verteilen
            acc_l += (weighted * pan_l[i]) >> 15;
            acc_r += (weighted * pan_r[i]) >> 15;

            // Phase weiterschalten (automatischer Wrap bei 2^32)
            v->phase[i] = p + dt;
        }

        // Hard-Clipping-Schutz für 16-Bit DAC-Ausgang
        if (acc_l > 32767) acc_l = 32767;
        else if (acc_l < -32768) acc_l = -32768;

        if (acc_r > 32767) acc_r = 32767;
        else if (acc_r < -32768) acc_r = -32768;

        out_l[n] = (int16_t)acc_l;
        out_r[n] = (int16_t)acc_r;
    }
}
```

```c
#include <stdint.h>

#define SUPERSAW_VOICES 7

static const int32_t kDetuneOffsets[SUPERSAW_VOICES] = {
    -7210, -4064, -1280, 0, 1280, 4064, 7210
};

static const int32_t kOscGainsQ15[SUPERSAW_VOICES] = {
    3900, 4300, 4700, 6968, 4700, 4300, 3900
};

typedef struct {
    uint32_t phase[SUPERSAW_VOICES];
    uint32_t inc[SUPERSAW_VOICES];
    uint32_t inv_dt[SUPERSAW_VOICES]; // Vorberechneter Kehrwert für PolyBLEP

    // Zustandsvariablen für 1-Pol-Hochpassfilter (DC-Blocker / Sub-Cut)
    int32_t hp_x1; // Vorheriges Eingangssample (Q15)
    int32_t hp_y1; // Vorheriges Ausgangssample (Q15)
    int32_t hp_alpha; // Filter-Koeffizient in Q15 (0 .. 32767)
} SupersawState;

// 1. DIVISION-FREIES POLYBLEP (keine 64-Bit-Division im Audio-Loop!)
static inline int32_t polyblep_fast_q15(uint32_t t, uint32_t dt, uint32_t inv_dt) {
    if (t < dt) {
        // u = (t / dt) * 32768  -->  ersetzt durch Multiplikation mit inv_dt!
        int32_t u = (int32_t)(((uint64_t)t * inv_dt) >> 32);
        int32_t u2 = (u * u) >> 15;
        return (u << 1) - u2 - 32768;
    }
    uint32_t dist_to_end = ~t;
    if (dist_to_end < dt) {
        int32_t u = -(int32_t)(((uint64_t)dist_to_end * inv_dt) >> 32);
        int32_t u2 = (u * u) >> 15;
        return u2 + (u << 1) + 32768;
    }
    return 0;
}

// 2. CONTROL-BLOCK UPDATE (wird nur alle 32 Samples aufgerufen!)
void supersaw_update_block(SupersawState *s, uint32_t base_inc, int32_t detune_q15) {
    for (int i = 0; i < SUPERSAW_VOICES; i++) {
        int64_t mod = (int64_t)kDetuneOffsets[i] * detune_q15;
        int64_t delta = ((int64_t)base_inc * mod) >> 31;
        uint32_t dt = (uint32_t)((int64_t)base_inc + delta);
        if (dt == 0) dt = 1;

        s->inc[i] = dt;
        // Kehrwert einmal pro 32 Samples berechnen: (2^47) / dt
        s->inv_dt[i] = (uint32_t)((1ULL << 47) / dt);
    }

    // 1-Pol-Hochpass auf Grundfrequenz abstimmen (wie beim JP-8000):
    // Grenzfrequenz f_c = Grundfrequenz => alpha ≈ 1 - (2 * pi * base_inc / 2^32)
    // In Q15: 2 * pi * 32768 ≈ 205887
    int32_t omega_q15 = (int32_t)(((uint64_t)base_inc * 205887ULL) >> 32);
    if (omega_q15 > 16384) omega_q15 = 16384; // Begrenzung bei sehr hohen Noten
    s->hp_alpha = 32768 - omega_q15;
}

// 3. AUDIO-LOOP (wird in eng_analog.c inside render() aufgerufen)
void supersaw_render_block(SupersawState *s, int32_t *out_buf, int n) {
    int32_t x1 = s->hp_x1;
    int32_t y1 = s->hp_y1;
    int32_t alpha = s->hp_alpha;

    for (int sample = 0; sample < n; sample++) {
        int32_t mix = 0;

        for (int i = 0; i < SUPERSAW_VOICES; i++) {
            uint32_t p = s->phase[i];
            uint32_t dt = s->inc[i];

            int32_t saw = (int32_t)(p >> 16) - 32768;
            saw -= polyblep_fast_q15(p, dt, s->inv_dt[i]);

            mix += (saw * kOscGainsQ15[i]) >> 15;
            s->phase[i] = p + dt;
        }

        // 1-Pol-Hochpass: y[n] = alpha * (y[n-1] + x[n] - x[n-1])
        // Räumt Sub-Bässe unter der Grundtonhöhe auf und sorgt für Club-Punch
        int32_t hp_out = (int32_t)(((int64_t)alpha * (y1 + mix - x1)) >> 15);
        x1 = mix;
        y1 = hp_out;

        // Auf Felucca-Voice-Buffer addieren (oder an den SVF-Filter weitergeben)
        out_buf[sample] += hp_out;
    }

    s->hp_x1 = x1;
    s->hp_y1 = y1;
}
```
